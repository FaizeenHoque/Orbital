STEP 6: Create server.js
  # In root directory, create server/server.js with this content:
  
  const express = require('express');
  const cors = require('cors');
  require('dotenv').config();
  const spherexApi = require('./spherex-api');

  const app = express();
  const PORT = process.env.PORT || 5000;

  app.use(cors());
  app.use(express.json());
  app.use(express.static('public'));
  app.use('/api/spherex', spherexApi);

  app.get('/health', (req, res) => {
    res.json({ status: 'ok', message: 'SPHEREx API running' });
  });

  app.listen(PORT, () => {
    console.log(`🌌 API running on http://localhost:${PORT}`);
  });

WHAT EACH FILE DOES
================================================================================

spherex-parser.js:
  INPUT: 1790441094847_spherex-example.html (your HTML table)
  OUTPUT: 
    - spherex-catalog.json (all observations as JSON array)
    - spherex-spatial-index.json (RA/Dec grid index for fast queries)
  
  Run once, then never again.

spherex-api.js:
  Implements these endpoints:
  - GET /observations?ra=X&dec=Y&radius=Z&band=BAND
  - POST /detect-moving-objects (compare two surveys)
  - GET /image/:obsId (fetch image from IRSA)
  - GET /stats (catalog statistics)
  - GET /search?q=TARGET (find by name)
  
  No modifications needed - copy as-is.

SkyExplorer.jsx:
  Main UI component with:
  - Sky coordinates input
  - Wavelength band selector
  - Survey comparison controls
  - Time-lapse slider
  - Moving objects display
  - Left/center/right panel layout
  
  No modifications needed - copy as-is.

CanvasSkyMap.jsx:
  Advanced visualization:
  - Draws starfield background
  - Shows observations as colored dots (wavelength-based)
  - Highlights moving objects with motion arrows
  - Interactive (zoom, pan, hover)
  
  Optional - use if you want advanced visuals.
